## Overview

Implement automatic update mechanism using `electron-updater` with GitHub releases as update server.

## Architecture

```
Update Flow:
1. App checks for updates on startup
2. Downloads in background
3. Notifies user when ready
4. Restarts to apply update
```

## Implementation

### 1. Install electron-updater

```bash
cd frontend/packages/desktop
npm install electron-updater
```

### 2. Main Process Update Handler (main/main.ts)

```typescript
import { autoUpdater } from 'electron-updater'

const setupUpdater = () => {
  autoUpdater.checkForUpdatesAndNotify()
  
  // Check for updates every 10 minutes
  setInterval(() => {
    autoUpdater.checkForUpdates()
  }, 10 * 60 * 1000)
}

app.on('ready', () => {
  setupUpdater()
  createWindow()
})

// Listen for update events
autoUpdater.on('update-available', () => {
  mainWindow?.webContents.send('update-available')
})

autoUpdater.on('update-downloaded', () => {
  mainWindow?.webContents.send('update-downloaded')
})
```

### 3. Preload API (preload/index.ts)

```typescript
interface ElectronAPI {
  // ... existing methods
  
  // Update methods
  onUpdateAvailable: (callback: () => void) => void
  onUpdateDownloaded: (callback: () => void) => void
  installUpdate: () => Promise<void>
}

const electronAPI: ElectronAPI = {
  // ... existing
  
  onUpdateAvailable: (callback: () => void) =>
    ipcRenderer.on('update-available', callback),
  
  onUpdateDownloaded: (callback: () => void) =>
    ipcRenderer.on('update-downloaded', callback),
  
  installUpdate: () => ipcRenderer.invoke('install-update'),
}
```

### 4. Add IPC Handler (main/main.ts)

```typescript
ipcMain.handle('install-update', async () => {
  autoUpdater.quitAndInstall()
})
```

### 5. React Component for Update Notification

```typescript
// src/components/UpdateNotification.tsx
import { useEffect, useState } from 'react'

export const UpdateNotification = () => {
  const [updateAvailable, setUpdateAvailable] = useState(false)
  const [updateDownloaded, setUpdateDownloaded] = useState(false)

  useEffect(() => {
    if (!window.electron) return

    window.electron.onUpdateAvailable(() => {
      setUpdateAvailable(true)
    })

    window.electron.onUpdateDownloaded(() => {
      setUpdateDownloaded(true)
      setUpdateAvailable(false)
    })
  }, [])

  if (!updateDownloaded) return null

  return (
    <div className="fixed bottom-4 right-4 bg-blue-600 text-white p-4 rounded">
      <p>Update downloaded - restart to apply</p>
      <button
        onClick={() => window.electron.installUpdate()}
        className="mt-2 px-4 py-2 bg-white text-blue-600 rounded"
      >
        Restart Now
      </button>
    </div>
  )
}
```

### 7. Release Process

#### Create GitHub Release:

```bash
# Tag the release
git tag v2.0.0

# Push tag
git push origin v2.0.0

# Build installers
npm run dist

# Create GitHub release with build artifacts
# 1. Go to GitHub → Releases → New Release
# 2. Select tag v2.0.0
# 3. Upload:
#    - GrowAgentAI Setup 2.0.0.exe
#    - GrowAgentAI 2.0.0.exe (portable)
#    - GrowAgentAI-2.0.0.dmg
#    - GrowAgentAI-2.0.0.AppImage
#    - latest.yml (generated by electron-builder)
```
